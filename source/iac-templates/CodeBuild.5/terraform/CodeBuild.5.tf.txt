# CodeBuild.5 — CodeBuild project environments should not have privileged mode enabled
#
# ASR remediation disables privileged mode on the CodeBuild project environment,
# removing the build container's access to all host devices.
# To apply this fix, set privileged_mode to false in the environment block.

resource "aws_codebuild_project" "{codeBuildProject}" {
  environment {
    privileged_mode = false
  }
}
