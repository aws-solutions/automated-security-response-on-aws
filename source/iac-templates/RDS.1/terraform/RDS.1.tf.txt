# RDS.1 — RDS snapshots should be private
#
# ASR remediation removes public access from RDS DB snapshots and cluster
# snapshots by revoking the "all" restore permission.
# To apply this fix, ensure shared_accounts does not include "all" on your
# aws_db_snapshot or aws_db_cluster_snapshot resources.

resource "aws_db_snapshot" "{dbSnapshot}" {
  shared_accounts = []
}
