# S3.14 — S3 general purpose buckets should have versioning enabled
#
# ASR remediation enables versioning on the bucket.
# To apply this fix, add an aws_s3_bucket_versioning resource for your bucket.

resource "aws_s3_bucket_versioning" "{bucketName}" {
  bucket = aws_s3_bucket.{bucketName}.id

  versioning_configuration {
    status = "Enabled"
  }
}
