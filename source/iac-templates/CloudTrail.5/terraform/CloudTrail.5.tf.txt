# CloudTrail.5 — CloudTrail trails should be integrated with Amazon CloudWatch Logs
#
# ASR remediation configures the CloudTrail trail to send logs to a CloudWatch Logs log group.
# To apply this fix, add or update the cloud_watch_logs_group_arn and cloud_watch_logs_role_arn
# arguments on your CloudTrail resource.
# NOTE: <LOG_GROUP> — your CloudWatch Logs log group for CloudTrail.
# NOTE: <CLOUDTRAIL_LOGS_ROLE> — an IAM role for CloudTrail-to-CloudWatch delivery.

resource "aws_cloudtrail" "{trailName}" {
  cloud_watch_logs_group_arn = "${<LOG_GROUP>}:*"
  cloud_watch_logs_role_arn  = "<CLOUDTRAIL_LOGS_ROLE>"
}
