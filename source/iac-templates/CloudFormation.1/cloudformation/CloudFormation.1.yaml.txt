# CloudFormation.1 — CloudFormation stacks should be integrated with SNS
#
# ASR remediation creates an SNS topic and configures the stack to send
# notifications to it.
# To apply this fix, add NotificationARNs to your CloudFormation stack resource.
# NOTE: <NOTIFICATION_TOPIC> — your SNS topic for stack notifications.

Resources:
  {stack}:
    Type: AWS::CloudFormation::Stack
    Properties:
      NotificationARNs:
        - <NOTIFICATION_TOPIC>
