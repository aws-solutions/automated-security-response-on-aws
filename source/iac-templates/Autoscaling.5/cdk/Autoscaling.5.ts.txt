// Autoscaling.5 — Launch configurations should not assign a public IP address to instances
//
// ASR remediation creates a new launch configuration with public IP address
// assignment disabled and updates the Auto Scaling group to use it.
// To apply this fix, set associatePublicIpAddress to false on your
// CfnLaunchConfiguration.

new autoscaling.CfnLaunchConfiguration(this, '{launchConfiguration}', {
  associatePublicIpAddress: false,
});
