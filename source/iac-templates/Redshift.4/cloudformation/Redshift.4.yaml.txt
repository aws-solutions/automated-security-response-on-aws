# Redshift.4 — Amazon Redshift clusters should have audit logging enabled
#
# ASR remediation enables audit logging on the Redshift cluster to an S3 bucket.
# To apply this fix, add LoggingProperties with your S3 bucket name on your Redshift cluster.
# NOTE: <AUDIT_LOGGING_BUCKET> — your S3 bucket for audit logs.

Resources:
  {redshiftCluster}:
    Type: AWS::Redshift::Cluster
    Properties:
      LoggingProperties:
        BucketName: <AUDIT_LOGGING_BUCKET>
        S3KeyPrefix: redshift-audit-logs/
