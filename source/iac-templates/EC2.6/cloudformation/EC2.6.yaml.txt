# EC2.6 — VPC flow logging should be enabled
#
# ASR remediation creates a CloudWatch Logs log group and enables VPC Flow Logs
# to capture rejected traffic.
# To apply this fix, add a CloudWatch Logs log group and a VPC Flow Log resource
# to your VPC definition.
# NOTE: <KMS_KEY> — your KMS key ARN for log group encryption.
# NOTE: {flowLogRole} — an IAM role that grants logs:CreateLogGroup,
# logs:CreateLogStream, and logs:PutLogEvents permissions.

Resources:
  {flowLogGroup}:
    Type: AWS::Logs::LogGroup
    Properties:
      LogGroupName: "VPCFlowLogs/{vpcId}"
      KmsKeyId: <KMS_KEY>

  {flowLog}:
    Type: AWS::EC2::FlowLog
    Properties:
      ResourceId: "{vpcId}"
      ResourceType: VPC
      TrafficType: REJECT
      LogDestinationType: cloud-watch-logs
      LogGroupName: !Ref {flowLogGroup}
      DeliverLogsPermissionArn: !GetAtt {flowLogRole}.Arn
