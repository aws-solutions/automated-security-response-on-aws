# S3.11 — S3 buckets should have event notifications enabled
#
# ASR remediation creates an SNS topic and configures event notifications
# on the S3 bucket to publish to that topic for key event types.
# To apply this fix, add an SNS topic and an aws_s3_bucket_notification
# resource with the desired event types.

data "aws_caller_identity" "current" {}
resource "aws_sns_topic" "bucket_notifications" {
  name = "SO0111-ASR-S3BucketNotifications"
}

resource "aws_sns_topic_policy" "bucket_notifications" {
  arn = aws_sns_topic.bucket_notifications.arn

  policy = jsonencode({
    Version = "2012-10-17"
    Statement = [
      {
        Sid       = "AllowS3Publish"
        Effect    = "Allow"
        Principal = { Service = "s3.amazonaws.com" }
        Action    = "sns:Publish"
        Resource  = aws_sns_topic.bucket_notifications.arn
        Condition = {
          ArnLike = {
            "aws:SourceArn" = aws_s3_bucket.{bucketName}.arn
          }
          StringEquals = {
            "aws:SourceAccount" = data.aws_caller_identity.current.account_id
          }
        }
      }
    ]
  })
}

resource "aws_s3_bucket_notification" "{bucketName}" {
  bucket = aws_s3_bucket.{bucketName}.id

  topic {
    topic_arn = aws_sns_topic.bucket_notifications.arn
    events = [
      "s3:ReducedRedundancyLostObject",
      "s3:ObjectCreated:*",
      "s3:ObjectRemoved:*",
      "s3:ObjectRestore:*",
      "s3:Replication:*",
      "s3:LifecycleExpiration:*",
      "s3:LifecycleTransition",
      "s3:IntelligentTiering",
      "s3:ObjectTagging:*",
      "s3:ObjectAcl:Put",
    ]
  }
}
