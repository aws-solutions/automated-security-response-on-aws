// SecretsManager.1 — Secrets Manager secrets should have automatic rotation enabled
//
// ASR remediation enables automatic rotation on the secret with a 90-day schedule.
// To apply this fix, add a rotation schedule with automaticallyAfter set to
// 90 days on your secret.
// NOTE: <ROTATION_FUNCTION> — your rotation Lambda function.

const secret = new secretsmanager.Secret(this, '{secret}');

secret.addRotationSchedule('RotationSchedule', {
  rotationLambda: <ROTATION_FUNCTION>,
  automaticallyAfter: cdk.Duration.days(90),
});
