# SecretsManager.1 — Secrets Manager secrets should have automatic rotation enabled
#
# ASR remediation enables automatic rotation on the secret with a 90-day schedule.
# To apply this fix, add a RotationSchedule resource with AutomaticallyAfterDays
# set to 90 for your secret.
# NOTE: <ROTATION_FUNCTION> — your rotation Lambda function ARN.

Resources:
  {secret}:
    Type: AWS::SecretsManager::Secret

  SecretRotationSchedule:
    Type: AWS::SecretsManager::RotationSchedule
    Properties:
      SecretId: !Ref {secret}
      RotationLambdaARN: <ROTATION_FUNCTION>
      RotationRules:
        AutomaticallyAfterDays: 90
