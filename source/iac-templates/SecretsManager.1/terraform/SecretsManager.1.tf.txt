# SecretsManager.1 — Secrets Manager secrets should have automatic rotation enabled
#
# ASR remediation enables automatic rotation on the secret with a 90-day schedule.
# To apply this fix, add an aws_secretsmanager_secret_rotation resource with
# automatically_after_days set to 90 for your secret.
# NOTE: <ROTATION_FUNCTION> — your rotation Lambda function ARN.

resource "aws_secretsmanager_secret_rotation" "{secret}" {
  secret_id           = aws_secretsmanager_secret.{secret}.id
  rotation_lambda_arn = "<ROTATION_FUNCTION>"

  rotation_rules {
    automatically_after_days = 90
  }
}
