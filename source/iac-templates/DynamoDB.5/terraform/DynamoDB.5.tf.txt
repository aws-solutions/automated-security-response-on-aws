# DynamoDB.5 — DynamoDB tables should be tagged
#
# ASR remediation tags the DynamoDB table with the required tag keys
# specified in the Security Hub control parameters.
# To apply this fix, add the required tags to your DynamoDB table resource.

resource "aws_dynamodb_table" "{tableName}" {
  tags = {
    "SO0111-ASR-DynamoDBTable" = ""
  }
}
