# S3.2 — S3 buckets should prohibit public read access
#
# ASR remediation enables the S3 bucket-level PublicAccessBlock configuration,
# blocking all public ACLs and public bucket policies for the bucket.
# To apply this fix, add an aws_s3_bucket_public_access_block resource
# with all four settings set to true.

resource "aws_s3_bucket_public_access_block" "{bucketName}" {
  bucket = aws_s3_bucket.{bucketName}.id

  block_public_acls       = true
  block_public_policy     = true
  ignore_public_acls      = true
  restrict_public_buckets = true
}
