# APIGateway.5 — API Gateway REST API cache data should be encrypted at rest
#
# ASR remediation enables cache data encryption on all method settings.
# To apply this fix, set cache_data_encrypted to true in your stage's
# method_settings block.

resource "aws_api_gateway_method_settings" "all" {
  rest_api_id = aws_api_gateway_rest_api.{ApiStage}.id
  stage_name  = aws_api_gateway_stage.{ApiStage}.stage_name
  method_path = "*/*"

  settings {
    cache_data_encrypted = true
  }
}
