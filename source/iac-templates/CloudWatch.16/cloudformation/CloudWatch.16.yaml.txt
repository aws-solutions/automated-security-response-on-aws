# CloudWatch.16 — CloudWatch log groups should be retained for a specified time period
#
# ASR remediation sets the retention period on the log group to 365 days.
# To apply this fix, add or update the RetentionInDays property on your log group resource.

Resources:
  {logGroup}:
    Type: AWS::Logs::LogGroup
    Properties:
      RetentionInDays: 365
