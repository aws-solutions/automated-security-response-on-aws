# ElastiCache.3 — ElastiCache replication groups should have automatic failover enabled
#
# ASR remediation enables automatic failover for the ElastiCache replication group.
# To apply this fix, set automatic_failover_enabled to true on your ElastiCache
# replication group.

resource "aws_elasticache_replication_group" "{replicationGroup}" {
  automatic_failover_enabled = true
}
