# ElastiCache.3 — ElastiCache replication groups should have automatic failover enabled
#
# ASR remediation enables automatic failover for the ElastiCache replication group.
# To apply this fix, set AutomaticFailoverEnabled to true on your ElastiCache
# replication group.

Resources:
  {replicationGroup}:
    Type: AWS::ElastiCache::ReplicationGroup
    Properties:
      AutomaticFailoverEnabled: true
