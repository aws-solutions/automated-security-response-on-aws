# RDS.5 — RDS DB instances should be configured with multiple Availability Zones
#
# ASR remediation enables Multi-AZ on the RDS DB instance by calling
# ModifyDBInstance with MultiAZ set to true.
# To apply this fix, set multi_az to true on your aws_db_instance resource.

resource "aws_db_instance" "{dbInstance}" {
  multi_az = true
}
