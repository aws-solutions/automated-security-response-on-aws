# ElastiCache.2 — ElastiCache clusters should have automatic minor version upgrades enabled
#
# ASR remediation enables automatic minor version upgrades for the ElastiCache cluster.
# To apply this fix, set AutoMinorVersionUpgrade to true on your ElastiCache cache cluster.

Resources:
  {cacheCluster}:
    Type: AWS::ElastiCache::CacheCluster
    Properties:
      AutoMinorVersionUpgrade: true
