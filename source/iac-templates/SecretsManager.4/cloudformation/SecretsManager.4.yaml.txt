# SecretsManager.4 — Secrets Manager secrets should be rotated within a specified number of days
#
# ASR remediation rotates the secret and sets its rotation schedule to 90 days.
# To apply this fix, add or update the RotationSchedule resource with
# AutomaticallyAfterDays set to 90 for your secret.
# NOTE: <ROTATION_FUNCTION> — your rotation Lambda function ARN.

Resources:
  {secret}:
    Type: AWS::SecretsManager::Secret

  SecretRotationSchedule:
    Type: AWS::SecretsManager::RotationSchedule
    Properties:
      SecretId: !Ref {secret}
      RotationLambdaARN: <ROTATION_FUNCTION>
      RotationRules:
        AutomaticallyAfterDays: 90
