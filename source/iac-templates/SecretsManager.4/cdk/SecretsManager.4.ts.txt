// SecretsManager.4 — Secrets Manager secrets should be rotated within a specified number of days
//
// ASR remediation rotates the secret and sets its rotation schedule to 90 days.
// To apply this fix, add or update the rotation schedule with automaticallyAfter
// set to 90 days on your secret.
// NOTE: <ROTATION_FUNCTION> — your rotation Lambda function.

const secret = new secretsmanager.Secret(this, '{secret}');

secret.addRotationSchedule('RotationSchedule', {
  rotationLambda: <ROTATION_FUNCTION>,
  automaticallyAfter: cdk.Duration.days(90),
});
