# SecretsManager.4 — Secrets Manager secrets should be rotated within a specified number of days
#
# ASR remediation rotates the secret and sets its rotation schedule to 90 days.
# To apply this fix, add or update the aws_secretsmanager_secret_rotation resource
# with automatically_after_days set to 90 for your secret.
# NOTE: <ROTATION_FUNCTION> — your rotation Lambda function ARN.

resource "aws_secretsmanager_secret_rotation" "{secret}" {
  secret_id           = aws_secretsmanager_secret.{secret}.id
  rotation_lambda_arn = "<ROTATION_FUNCTION>"

  rotation_rules {
    automatically_after_days = 90
  }
}
