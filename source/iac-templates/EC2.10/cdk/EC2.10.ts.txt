// EC2.10 — Amazon EC2 should be configured to use VPC endpoints
//
// ASR remediation creates and attaches an interface VPC endpoint for the
// specified AWS service to the VPC.
// To apply this fix, add an InterfaceVpcEndpoint construct with the service
// name and VPC reference in your CDK stack.
// NOTE: <VPC> — your ec2.Vpc construct. The subnets property uses
// PRIVATE_WITH_EGRESS to auto-select subnets; adjust if your VPC uses a
// different subnet layout.

new ec2.InterfaceVpcEndpoint(this, '{vpcEndpoint}', {
  vpc: <VPC>, // {vpcId}
  service: new ec2.InterfaceVpcEndpointService('com.amazonaws.{region}.ec2'),
  subnets: { subnetType: ec2.SubnetType.PRIVATE_WITH_EGRESS },
});
