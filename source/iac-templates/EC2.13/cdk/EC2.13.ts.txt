// EC2.13 — Security groups should not allow ingress from 0.0.0.0/0 or ::/0 to port 22
//
// ASR remediation removes inbound rules that allow unrestricted access (0.0.0.0/0
// or ::/0) to administrative ports (SSH 22, RDP 3389) from the security group.
// To apply this fix, restrict the peer on your SSH and RDP ingress rules to
// trusted IP ranges instead of Peer.anyIpv4() or Peer.anyIpv6().
// NOTE: <VPC> — your ec2.Vpc construct.

const securityGroup = new ec2.SecurityGroup(this, '{securityGroup}', {
  vpc: <VPC>,
  description: 'Security group with restricted administrative access',
});

securityGroup.addIngressRule(
  ec2.Peer.ipv4('10.0.0.0/8'),
  ec2.Port.tcp(22),
  'SSH from trusted network',
);

securityGroup.addIngressRule(
  ec2.Peer.ipv4('10.0.0.0/8'),
  ec2.Port.tcp(3389),
  'RDP from trusted network',
);
