# CloudFront.12 — CloudFront distributions should not point to non-existent S3 origins
#
# ASR remediation sets the origin domain to a safe placeholder value and disables the distribution
# to prevent a potential malicious takeover of an orphaned S3 origin.
# To apply this fix, update the origin DomainName to a valid origin and set Enabled to false on
# your CloudFront distribution.
# NOTE: <ORIGIN_DOMAIN> — your actual origin domain.

Resources:
  {distribution}:
    Type: AWS::CloudFront::Distribution
    Properties:
      DistributionConfig:
        Enabled: false
        Origins:
          - DomainName: <ORIGIN_DOMAIN>
            Id: SafePlaceholderOrigin
            CustomOriginConfig:
              HTTPPort: 80
              HTTPSPort: 443
              OriginProtocolPolicy: http-only
              OriginSSLProtocols:
                - TLSv1.2
