# CloudFront.12 — CloudFront distributions should not point to non-existent S3 origins
#
# ASR remediation sets the origin domain to a safe placeholder value and disables the distribution
# to prevent a potential malicious takeover of an orphaned S3 origin.
# To apply this fix, update the origin domain_name to a valid origin and set enabled to false on
# your CloudFront distribution.
# NOTE: <ORIGIN_DOMAIN> — your actual origin domain.

resource "aws_cloudfront_distribution" "{distribution}" {
  enabled = false

  origin {
    domain_name = "<ORIGIN_DOMAIN>"
    origin_id   = "SafePlaceholderOrigin"

    custom_origin_config {
      http_port              = 80
      https_port             = 443
      origin_protocol_policy = "http-only"
      origin_ssl_protocols   = ["TLSv1.2"]
    }
  }
}
