# CloudFront.12 — CloudFront distributions should not point to non-existent S3 origins
#
# ASR remediation replaces the domain of each S3 origin whose bucket no longer exists with a
# safe placeholder value, to prevent a potential malicious takeover of an orphaned S3 origin.
# Other origins and the distribution's enabled state are not changed.
# To apply this fix, point each affected origin at a valid origin, or remove it from your
# CloudFront distribution.
# NOTE: <ORIGIN_DOMAIN> — your actual origin domain.

resource "aws_cloudfront_distribution" "{distribution}" {
  origin {
    domain_name = "<ORIGIN_DOMAIN>"
    origin_id   = "SafePlaceholderOrigin"

    custom_origin_config {
      http_port              = 80
      https_port             = 443
      origin_protocol_policy = "http-only"
      origin_ssl_protocols   = ["TLSv1.2"]
    }
  }
}
