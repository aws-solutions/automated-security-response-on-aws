# SNS.1 — SNS topics should be encrypted at rest using AWS KMS
#
# ASR remediation enables encryption at rest on the SNS topic using a KMS key.
# To apply this fix, add or update the kms_master_key_id argument on your SNS topic resource.
# NOTE: <KMS_KEY> — your KMS key for topic encryption.

resource "aws_sns_topic" "{snsTopic}" {
  kms_master_key_id = "<KMS_KEY>"
}
