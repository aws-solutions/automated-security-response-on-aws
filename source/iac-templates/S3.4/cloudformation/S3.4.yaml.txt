# S3.4 — S3 buckets should have server-side encryption enabled
#
# ASR remediation enables default server-side encryption on the S3 bucket
# using AES-256 (SSE-S3) with an S3 Bucket Key.
# To apply this fix, add a BucketEncryption configuration to your S3 bucket
# with SSEAlgorithm set to AES256 and BucketKeyEnabled set to true.

Resources:
  {bucketName}:
    Type: AWS::S3::Bucket
    Properties:
      BucketEncryption:
        ServerSideEncryptionConfiguration:
          - ServerSideEncryptionByDefault:
              SSEAlgorithm: AES256
            BucketKeyEnabled: true
