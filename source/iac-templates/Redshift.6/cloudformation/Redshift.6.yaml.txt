# Redshift.6 — Amazon Redshift should have automatic upgrades to major versions enabled
#
# ASR remediation enables automatic version upgrade on the Redshift cluster.
# To apply this fix, set AllowVersionUpgrade to true on your Redshift cluster.

Resources:
  {redshiftCluster}:
    Type: AWS::Redshift::Cluster
    Properties:
      AllowVersionUpgrade: true
