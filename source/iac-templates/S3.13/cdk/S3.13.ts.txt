// S3.13 — S3 buckets should have lifecycle policies configured
//
// ASR remediation sets an example lifecycle policy that transitions objects
// larger than 128 KB to Intelligent-Tiering after 30 days.
// To apply this fix, add a lifecycleRules entry to your S3 bucket
// with a transition to your desired storage class.

new s3.Bucket(this, '{bucketName}', {
  lifecycleRules: [
    {
      id: 'S3.13 Remediation Example',
      enabled: true,
      objectSizeGreaterThan: 131072,
      transitions: [
        {
          storageClass: s3.StorageClass.INTELLIGENT_TIERING,
          transitionAfter: cdk.Duration.days(30),
        },
      ],
    },
  ],
});
